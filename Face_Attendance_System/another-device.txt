━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Node.js  (v18 or higher)
     Download: https://nodejs.org/

  2. Python  (v3.10 – 3.13)
     Download: https://www.python.org/downloads/
     ⚠ During install, check "Add Python to PATH"

  3. MongoDB Community Server
     Download: https://www.mongodb.com/try/download/community
     Make sure the MongoDB service is running before starting the backend.

  4. Git  (optional, to clone the repo)
     Download: https://git-scm.com/

  5. A working webcam (built-in or USB)

  STEP-BY-STEP SETUP COMMANDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Open THREE separate terminal windows for the three services below.

  ── STEP 1: Navigate to the project root ───────────────────────────────────

    cd Face_Attendance_System


  ── STEP 2: Backend setup & start ──────────────────────────────────────────

    cd backend
    npm install
    npm run dev

    Backend will start at: http://localhost:5000
    You should see:  [MongoDB] Connected successfully
                     [Server] Running on http://localhost:5000

    ⚠ MongoDB must be running before this step.
      Windows: open Services → start "MongoDB Server"
      Or run:  net start MongoDB


  ── STEP 3: Frontend setup & start (new terminal) ──────────────────────────

    cd frontend
    npm install
    npm run dev

    Dashboard will open at: http://localhost:5173


  ── STEP 4: Python virtual environment & dependencies ──────────────────────

    (Run from the project root: Face_Attendance_System/)

    # Create virtual environment one level UP from the project folder
    python -m venv .venv

    # Activate it
    # Windows PowerShell:
    .venv\Scripts\Activate.ps1

    # Windows CMD:
    .venv\Scripts\activate.bat

    # macOS / Linux:
    source .venv/bin/activate

    # Install dlib-bin FIRST (pre-built, no CMake required)
    pip install dlib-bin

    # Install the rest of the Python dependencies
    pip install -r Face_Attendance_System/python-service/requirements.txt
    pip install setuptools

    # If you get a "pkg_resources" error, patch face_recognition_models:
    # Open: .venv\Lib\site-packages\face_recognition_models\__init__.py
    # Remove or comment out the line:  import pkg_resources


  ── STEP 5: Start the Python face-recognition service (new terminal) ───────

    # Make sure the venv is activated (see Step 4)
    cd Face_Attendance_System/python-service

    python face_recognition_service.py

    You should see:
      [Socket] Connected to Node backend ✓
      [Info] Encoded N known face(s): [...]
      [Info] Face-recognition service started. Press 'q' in the OpenCV window to quit.

  ADDING EMPLOYEES (TRAINING IMAGES)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Place each employee's face photo inside:
    python-service/Training images/

  File naming:  <EmployeeName>.jpg   (e.g. John.jpg, Alice.jpg)

  After adding photos, restart the Python service (Step 5) so the new
  face encodings are loaded.

  Alternatively, use the "Add Employee" button on the dashboard — it uploads
  the photo automatically and saves it to the Training images folder.
  Still restart the Python service afterwards.

  TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Problem: "dlib" fails to install / CMake error
  Fix:     Install dlib-bin instead of dlib:
             pip install dlib-bin

  Problem: "pkg_resources not found" error from face_recognition_models
  Fix:     Edit .venv\Lib\site-packages\face_recognition_models\__init__.py
           Remove the line: import pkg_resources

  Problem: MongoDB connection failed
  Fix:     Start MongoDB service:  net start MongoDB  (Windows)
           Or install and start MongoDB Community Server.

  Problem: Python service says "Could not connect to backend"
  Fix:     Make sure the backend (Terminal 1) is running BEFORE starting
           the Python service.

  Problem: Webcam not opening / black screen
  Fix:     Check that no other app is using the webcam.
           Try changing CAMERA_INDEX = 0 to 1 in face_recognition_service.py

  Problem: ExecutionPolicy error when activating venv on Windows PowerShell
  Fix:     Run:  Set-ExecutionPolicy -Scope CurrentUser RemoteSigned
